cmake_minimum_required(VERSION 3.23)
project(spud LANGUAGES C CXX)
include(FetchContent)
include(GNUInstallDirs)
# Downstream-owned packaging for the unmodified snapshot; see SOURCE.md.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Honor the owning Xmake target's CRT choice, including MT debug builds.
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
option(SPUD_DETOUR_TRACING "Enable SPUD tracing ABI" OFF)
option(SPUD_NO_LTO "Disable link-time optimization" ON)
option(SPUD_BUILD_TESTS "Build vendor tests (not included in snapshot)" OFF)
include(SelectArchitecture.cmake)
spud_select_architecture(SPUD_PROCESSOR "${CMAKE_SYSTEM_PROCESSOR}" "${CMAKE_OSX_ARCHITECTURES}")
if(SPUD_PROCESSOR STREQUAL "arm64")
    set(SPUD_ARM ON)
else()
    set(SPUD_ARM OFF)
endif()
option(SPUD_AARCH64_SUPPORT "Include ARM64 relocator" ${SPUD_ARM})
if(SPUD_BUILD_TESTS)
    message(FATAL_ERROR "Snapshot has no vendor tests; use the repository relocation fixture")
endif()
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zydis GIT_REPOSITORY https://github.com/zyantific/zydis.git GIT_TAG v4.1.0)
FetchContent_MakeAvailable(zydis)
if(SPUD_AARCH64_SUPPORT)
    set(CAPSTONE_BUILD_STATIC ON CACHE BOOL "" FORCE)
    set(CAPSTONE_BUILD_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
    set(CAPSTONE_BUILD_MACOS_THIN OFF CACHE BOOL "" FORCE)
    set(CAPSTONE_ARCHITECTURE_DEFAULT OFF CACHE BOOL "" FORCE)
    set(CAPSTONE_AARCH64_SUPPORT ON CACHE BOOL "" FORCE)
    FetchContent_Declare(capstone GIT_REPOSITORY https://github.com/capstone-engine/capstone.git GIT_TAG 6.0.0-Alpha5)
    FetchContent_MakeAvailable(capstone)
endif()
add_library(spud STATIC
    src/utils.cc
    src/detour/detour.cc src/detour/chain.cc src/detour/control_flow.cc
    src/detour/pointer.cc src/detour/import.cc src/detour/x86_64/x86_64.cc
    src/memory/protection.cc src/memory/executable_pool.cc src/memory/safe_patch.cc
    src/memory/unwind.cc src/memory/unwind/windows.cc
    src/memory/unwind/windows_handler.cc src/memory/unwind/apple.cc
    src/memory/unwind/compact.cc src/memory/unwind/dwarf.cc src/memory/unwind/linux.cc
    src/signature/signature.cc)
target_include_directories(spud PUBLIC include PRIVATE src)
target_link_libraries(spud PRIVATE Zydis)
target_compile_definitions(spud PRIVATE NOMINMAX=1)
target_compile_definitions(spud PUBLIC SPUD_DETOUR_TRACING=$<BOOL:${SPUD_DETOUR_TRACING}>)
if(SPUD_AARCH64_SUPPORT)
    target_sources(spud PRIVATE src/detour/aarch64/aarch64.cc)
    target_compile_definitions(spud PUBLIC SPUD_AARCH64_SUPPORT=1)
    target_link_libraries(spud PRIVATE capstone)
endif()
if(SPUD_ARM)
    target_sources(spud PRIVATE src/signature/signature_neon.cc)
else()
    # Kernels carry per-function ISA attributes. Keep the TU baseline unchanged.
    target_sources(spud PRIVATE src/signature/signature_sse.cc src/signature/signature_avx2.cc)
    if(MSVC AND CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
        enable_language(ASM_MASM)
        set(SPUD_CONTEXT_ASM src/detour/x86_64/detour_context_support.masm)
        set_source_files_properties(${SPUD_CONTEXT_ASM} PROPERTIES LANGUAGE ASM_MASM)
        target_sources(spud PRIVATE ${SPUD_CONTEXT_ASM})
    endif()
endif()
if(MSVC)
    target_compile_options(spud PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/permissive->" "$<$<COMPILE_LANGUAGE:CXX>:/W4>")
endif()
if(NOT SPUD_NO_LTO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT SPUD_IPO_SUPPORTED)
    if(SPUD_IPO_SUPPORTED)
        set_property(TARGET spud PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
    endif()
endif()
install(TARGETS spud Zydis Zycore ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(SPUD_AARCH64_SUPPORT)
    install(TARGETS capstone ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
